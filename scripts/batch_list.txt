# RL_infra 批实验清单（scripts/run_batch.sh 读它）
# 格式： 实验名 | 额外命令行参数...
# · 实验名 = `main.py --name`，也是产物目录 runs/<name>/，跑完会写 .done
# · 空行与 # 开头的行为注释
# · 双卡并发（--mode par）时按顺序轮流钉到 cuda:0 / cuda:1
#
# ══════════════════════════════════════════════════════════════════
# 国庆批次（2026-09-30 定稿）
#
# 🔴 为什么必须加长：300 回合那批（见 eval_summary.csv）显示三档 lr 的独立测试
#    覆盖率落在 70.6~80.4%，**排序与训练最优不一致** ⇒ 300 回合信噪比不足以
#    分辨 lr，差异多半来自噪声。复跑必须上到 2000 回合。
#
# 🔴 一次只改一个变量：lr 组固定 seed=2024；seed 组固定 lr=5e-4（300 回合测试赢家）。
# ══════════════════════════════════════════════════════════════════

# ── 第 1 组：lr 三档 × 2000 回合（seed 固定 2024）──
lr2k_1e3_s2024 | --episodes 2000 --lr 0.001  --seed 2024 --parallel
lr2k_5e4_s2024 | --episodes 2000 --lr 0.0005 --seed 2024 --parallel
lr2k_1e4_s2024 | --episodes 2000 --lr 0.0001 --seed 2024 --parallel

# ── 第 2 组：seed 方差（lr 固定 5e-4 = 300 回合测试赢家）──
lr2k_5e4_s7  | --episodes 2000 --lr 0.0005 --seed 7  --parallel
lr2k_5e4_s13 | --episodes 2000 --lr 0.0005 --seed 13 --parallel

# ── 第 3 组：泛化对照（同配置，仅评估换地图）──
#    加 --eval-random-maps ⇒ 贪心评估每轮换地图，测的是泛化而不是"记住那张图"
lr2k_5e4_rand | --episodes 2000 --lr 0.0005 --seed 2024 --parallel --eval-random-maps
